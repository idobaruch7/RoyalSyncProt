emit שלב 1 — on_player_action

@socketio.on('player_action')
def on_player_action(data):
    if not current_game or not game_active:      # מתעלמים מ-events תועים (לובי, paused)
        return
    action = data.get('action', '')             # 'fold' | 'check' | 'call' | 'raise'
    amount = int(data.get('amount', 0))          # משמעותי רק עבור 'raise'

ה-handler מעביר את _request_sid() — ה-socket id של מי ששלח — הלאה למטה כזהות של הפועל. ה-sid הזה הוא איך ה-engine מאוחר יותר בודק "זה באמת התור שלך".

שלב 2 — _apply_and_advance (המנצח)

def _apply_and_advance(sid: str, action: str, amount: int):
    if not current_game:
        return

    actor = current_game.current_player()                     # A: זוכרים מי פועל לפני המוטציה
    _, event = current_game.apply_action(sid, action, amount) # B: validate + mutate, מקבלים event string
    _sync_all_game_player_chips()                             # C: צ'יפים מה-engine -> session_players
    _broadcast_game_state()                                   # D: דוחפים מצב ציבורי לכולם

    if event == 'invalid_action':                             # E
        session_id = sid_to_session.get(sid)
        info = session_players.get(session_id) if session_id else None
        if info and info.get('sid'):
            socketio.emit('action_error',
                {'message': current_game.last_action_error or 'Illegal action.'},
                to=info['sid'])
        _notify_current_player()                              # מבקשים שוב מאותו שחקן
        return

    if event == 'not_your_turn':                              # F: זורקים בשקט
        return

    if actor is not None:                                     # G: אומרים לכל המסכים מה קרה
        _emit_action_event(actor, action.lower(),
                           all_in=(actor.chips == 0 and not actor.folded))

    if event == 'game_over':                                  # H: שחקן אחרון שנשאר / showdown
        _broadcast_hand_over()
    elif event in ('continue', 'street_end'):                 # I: ממשיכים את מחזור התורות
        _process_automatic_turns()
  1. A — snapshot של הפועל קודם. אחרי apply_action ה-"current player" כבר התקדם, אז אנחנו תופסים את actor עכשיו כדי לתאר את המהלך מאוחר יותר ב-G.
  2. B — קריאה אחת עושה validate + mutate ומחזירה event string שאומר למנצח מה לעשות הלאה. זה הציר של כל העיצוב.
  3. C — sync של צ'יפים. במהלך יד אובייקט ה-Game אחראי על ספירת הצ'יפים; זה מעתיק אותם חזרה ל-session_players כדי שה-snapshots של lobby/queue יישארו נכונים.
  4. D — broadcast של המצב הציבורי ללא תנאי, אפילו על פעולה לא חוקית, כדי שהמסכים אף פעם לא יצאו מסנכרון.
  5. E — פעולה לא חוקית: שולחים action_error רק ל-socket הבודד הזה (to=info['sid']), ואז _notify_current_player() שולח מחדש את prompt ה-your_turn לאותו אדם. התור לא מתקדם.
  6. F — לא התור שלך: מישהו לחץ פעמיים או client מיושן שלח. זורקים בשקט — D כבר סנכרן אותם מחדש.
  7. G — action_event הוא ה-toast הקוסמטי "Alice raised to 40 / Bob folded" שכל מסך מציג. נפרד מ-game_state כי זה event, לא state.
  8. H / I — הסתעפות לפי ה-event: או שהיד הסתיימה (game_over) או שהמחזור ממשיך ואנחנו מעבירים ל-_process_automatic_turns().
למה ערכי ההחזרה חשובים. _apply_and_advance אף פעם לא בודק את השדות הפנימיים של המשחק כדי להחליט מה לעשות — הוא מגיב רק ל-event string. ה-engine ושכבת ה-socket נשארים מנותקים (decoupled).

    self.last_action_error = ''

תנועת צ'יפים — _commit_bet

def _commit_bet(self, player, amount):
    if amount <= 0: return
    player.chips    -= amount
    player.round_bet += amount     # תרומה בסבב ה-betting הזה (מתאפס בכל street)
    player.bet      += amount      # תרומה בכל היד (עבור side pots)
    self.pot        += amount

_emit_action_event — ה-toast של "מה קרה עכשיו"

def _emit_action_event(player, action, all_in=False):
    if not player or not action: return
    if action in ('call', 'raise'):
        amount = int(getattr(player, 'round_bet', 0) or 0)   # מציגים את הסכום שהוא עכשיו בפנים עליו
    else:
        amount = 0
    socketio.emit('action_event', {
        'actor': player.nickname, 'action': action,
        'amount': amount, 'all_in': bool(all_in),
    })

def _notify_current_player():
    if not current_game: return
    player = current_game.current_player()
    if player is None or player.sid is None:      # בוט, או אדם offline -> אין למי לשלוח prompt
        return
    socketio.emit('your_turn', {
        **current_game.legal_actions_for(player),  # אילו כפתורים להפעיל, min/max raise
        'big_blind': current_game.big_blind,
        'pot': current_game.pot,
    }, to=player.sid)                              # socket אחד בלבד

שלב 5 — _process_automatic_turns (למה השולחן אף פעם לא נתקע)

def _process_automatic_turns():
    while current_game and current_game.state.value not in ('waiting', 'showdown'):
        player = current_game.current_player()
        if player is None:
            return

        if isinstance(player, BotPlayer):
            _schedule_bot_turn(player)            # מעבירים ל-background task, ואז עוצרים את ה-loop
            return

        if isinstance(player, HumanPlayer) and not player.is_connected:
            call_amount = current_game.current_bet - getattr(player, 'round_bet', 0)
            action = 'check' if call_amount <= 0 else 'fold'   # ברירת מחדל ל-offline: check אם חינם, אחרת fold
            _, event = current_game.apply_action(None, action, 0)
            _emit_action_event(player, action, all_in=(player.chips == 0 and not player.folded))
            _sync_all_game_player_chips()
            _broadcast_game_state()
            if event == 'game_over':
                _broadcast_hand_over()
                return
            continue                              # loop שוב עבור השחקן הבא

        _notify_current_player()                  # אדם מחובר: שולחים לו prompt ומחכים
        return
  1. אדם מחובר בתור ← שולחים your_turn, יוצאים מה-loop, מחכים ל-player_action שלו.
  2. אדם offline בתור ← auto check-or-fold מיד, broadcast, ו-continue לשחקן הבא באותו loop.
  3. בוט בתור ← _schedule_bot_turn ו-return. ה-loop מתחדש מאוחר יותר, מתוך ה-background task של הבוט.

שלב 6 — המעקף של הבוט: _schedule_bot_turn / _run_bot_turn

_bot_action_pending = False   # guard ברמת המודול

def _schedule_bot_turn(player):
    global _bot_action_pending
    if _bot_action_pending:            # תור בוט כבר בטיסה -> לא לערום עוד אחד
        return
    _bot_action_pending = True
    socketio.start_background_task(_run_bot_turn, player)   # non-blocking; השרת ממשיך לשרת events
def _run_bot_turn(player):
    global _bot_action_pending
    try:
        socketio.sleep(random.uniform(0.7, 1.5))     # sleep async-mode-aware -> השהיית "חשיבה"

        if not current_game or current_game.state.value in ('waiting', 'showdown'):
            return
        if current_game.current_player() is not player:   # העולם התקדם בזמן שישנו
            return

        game_state = { **current_game.legal_actions_for(player),
                       'pot': current_game.pot,
                       'community_cards_objects': current_game.community_cards }
        decision = player.get_action(game_state) or {}
        act = (decision.get('action') or '').strip().lower()
        amount = int(decision.get('amount', 0) or 0)
        _, event = current_game.apply_action(None, act, amount)

        if event in ('invalid_action', 'not_your_turn', 'error'):   # הבוט בחר משהו לא חוקי
            act, amount = 'fold', 0
            _, event = current_game.apply_action(None, act, amount)

        _emit_action_event(player, act, all_in=(player.chips == 0 and not player.folded))
        _sync_all_game_player_chips()
        _broadcast_game_state()
        if event == 'game_over':
            _broadcast_hand_over()
            return
    finally:
        _bot_action_pending = False        # תמיד מנקים את ה-guard, גם על exception

    _process_automatic_turns()             # ממשיכים את המחזור (השחקן הבא יכול להיות בוט / אדם אחר)
אמור את זה בצורה קונקרטית (שבוע 2 מבקש את זה): קודם כול, הבוט "חושב" מתוך start_background_task ולא מתוך ה-handler — כך _apply_and_advance חוזר מיד ולא מחזיק את מהלך השחקן שהתחיל אותו. socketio.sleep() הוא async-mode-aware: תחת eventlet/gevent הוא מוותר (yields) ל-event loop כך שההשהיה של 0.7–1.5 שנ' לא חוסמת אף אחד, ותחת threading הוא נופל ל-time.sleep על ה-thread של ה-task בלבד. time.sleep() ישיר בתוך handler היה מחזיק את עיבוד ה-events של אותו client עד שהוא מסתיים — socketio.sleep בתוך background task הוא הדפוס הנכון והנייד.

ממה _bot_action_pending מגן: בלעדיו, טריגר שני מהיר (למשל disconnect ששולח _process_automatic_turns בזמן ש-task של בוט באמצע sleep) היה יכול לתזמן task שני לאותו בוט. שניהם מתעוררים, שניהם קוראים ל-apply_action לאותו מושב — השני רואה to_act שהשתנה ואז או פועל פעמיים או משבש את סדר התורות. ה-guard הופך את "תור בוט בעיצומו" ל-mutex. הבדיקה החוזרת current_player() is not player אחרי ה-sleep היא קו ההגנה השני.

תתאמן עד שזה שוטף:

"השחקן לוחץ Call — הדפדפן שולח (emit) player_action. on_player_action מגן וקורא ל-_apply_and_advance, שזוכר את הפועל וקורא ל-apply_action על ה-engine. ה-engine עושה validate — sid נכון, מהלך חוקי — ואז mutate: מעביר צ'יפים ל-pot, מוציא (pop) את הפועל מ-to_act, ומחזיר event: continue, game_over, או error. _apply_and_advance אז עושה broadcast — מסנכרן צ'יפים, שולח game_state לכולם, שולח action_event toast — ואז קורא או ל-_broadcast_hand_over או ל-_process_automatic_turns, ששולח prompt לאדם הבא, משחק אוטומטית עבור offline, או מתזמן בוט ב-background task non-blocking שחוזר בלולאה לאותו מחזור."
